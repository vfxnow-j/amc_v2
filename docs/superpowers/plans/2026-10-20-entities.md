# Entities Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let VFXnow track gear it stores for other owners ("entities") in separate databases, with an Inventory module, a Settings → Entities page, per-user access, and a Switch Entity menu — without that gear ever touching VFXnow's numbers.

**Architecture:** A registry (`Entity`, `UserEntity`) lives in the main database. Each entity gets its own Postgres database `vfxnow_ent_<slug>` with a small dedicated Prisma schema and generated client (`src/generated/entity-prisma`). Entity screens live under `/entity/[slug]/...` outside the `(shell)` group, behind a per-request grant check. Main code never imports the entity client; entity code reaches the main database only through `src/lib/entities/main-lookup.ts`.

**Tech Stack:** Next.js (read `node_modules/next/dist/docs/` before writing route/layout code — this version differs from training data), Prisma 7 with `@prisma/adapter-pg`, Postgres, `tsx --test` (node:test).

Spec: `docs/superpowers/specs/2026-10-20-entities-design.md`.

## Global Constraints

- Entity databases are named `vfxnow_ent_[a-z0-9_]+`; the entity client must refuse any other name and must refuse v1's database and `vfxnow_amc_v2`.
- `prisma db push` is **never** run against an entity database. Entity schema changes ship as numbered SQL files applied by `scripts/entities-migrate.ts`.
- `Movement` rows are append-only: enforced by a database trigger, not just by convention.
- Entity slug is immutable after creation (it is the database name).
- Entities are deactivated, never deleted. Nothing in this plan drops a database except the rollback of a failed provisioning.
- Only `SUPER_ADMIN` creates/edits entities and grants access. `FLOW_USER` can never be granted entity access. `VIEWER` is read-only inside an entity; `STAFF` and above can write.
- The active entity comes from the URL and is re-checked against grants on every request. Never trust a cookie.
- No code under `src/` outside `src/lib/entities/`, `src/app/entity/`, `src/components/entities/`, and `src/lib/actions/entities*.ts` may import `@/generated/entity-prisma` (enforced by Task 10).
- Never push `.env`, `backups/`, or `src/generated/`. Check `git diff --cached --stat` before every commit — staged renames/deletions ride along.
- Commit messages on this project are prose explaining decisions, ending with `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.
- Spec clarification (applied in Task 1): `RETURNED_TO_OWNER` is reached by a `RELEASE` with `finalToOwner = true`; the spec lists only four movement types.

```ts
export const ENTITY_DB_RE = /^vfxnow_ent_[a-z0-9_]{1,40}$/

export function slugify(name: string): string {
  return name
    .normalize('NFKD')
    .replace(/[̀-ͯ]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .slice(0, 40)
    .replace(/-+$/g, '')
}

export function dbNameFor(slug: string): string {
  return `vfxnow_ent_${slug.replace(/-/g, '_')}`
}
```

```ts
export type EntityRole = 'SUPER_ADMIN' | 'ADMIN' | 'STAFF' | 'VIEWER' | 'FLOW_USER'

export type AccessUser = {
  role: EntityRole
  entityAccess: boolean
  entityScope: 'ALL' | 'SELECTED'
  grantedEntityIds: string[]
}

export function canEnterEntities(u: AccessUser): boolean {
  if (u.role === 'FLOW_USER') return false
  return u.role === 'SUPER_ADMIN' || u.entityAccess
}

/** The entities (of `allIds`, i.e. the active ones) this user may open. */
export function entityIdsFor(u: AccessUser, allIds: string[]): string[] {
  if (!canEnterEntities(u)) return []
  if (u.role === 'SUPER_ADMIN' || u.entityScope === 'ALL') return allIds
  const granted = new Set(u.grantedEntityIds)
  return allIds.filter((id) => granted.has(id))
}

export function canWriteInEntity(role: EntityRole): boolean {
  return role === 'SUPER_ADMIN' || role === 'ADMIN' || role === 'STAFF'
}
```

```ts
/**
 * The entity client's counterpart to `assertV2Database`: an entity database
 * and nothing else. The main database and v1 are refused by name shape alone —
 * neither starts `vfxnow_ent_` — so there is no list of forbidden names to
 * keep current.
 */
export function assertEntityDatabase(url: string | undefined): string {
  if (!url) throw new Error("DATABASE_URL is not set");

  let name: string;
  try {
    name = decodeURIComponent(new URL(url).pathname).replace(/^\//, "");
  } catch {
    throw new Error("DATABASE_URL is not a valid connection string");
  }

  if (!/^vfxnow_ent_[a-z0-9_]{1,40}$/.test(name)) {
    throw new Error(
      `Refusing to connect to the database "${name}" as an entity database.\n` +
        `Entity databases are named vfxnow_ent_<slug>.`,
    );
  }
  return url;
}
```

```ts
import { assertEntityDatabase } from "@/lib/db-guard";
import { ENTITY_DB_RE } from "@/lib/entities/naming";
import { PrismaClient } from "@/generated/entity-prisma/client";
import { PrismaPg } from "@prisma/adapter-pg";
import { Pool } from "pg";

export type EntityDb = PrismaClient;

const globalForEntities = globalThis as unknown as {
  entityClients: Map<string, PrismaClient> | undefined;
};
const clients = (globalForEntities.entityClients ??= new Map());

/** The main DATABASE_URL with its database swapped for an entity's. */
export function entityUrl(dbName: string): string {
  if (!ENTITY_DB_RE.test(dbName)) throw new Error(`"${dbName}" is not an entity database name`);
  const base = process.env.DATABASE_URL;
  if (!base) throw new Error("DATABASE_URL is not set");
  const url = new URL(base);
  url.pathname = `/${dbName}`;
  return url.toString();
}

/** One pooled client per entity, kept on globalThis so hot reloads reuse it. */
export function entityDb(dbName: string): EntityDb {
  const existing = clients.get(dbName);
  if (existing) return existing;
  const connectionString = assertEntityDatabase(entityUrl(dbName));
  const client = new PrismaClient({ adapter: new PrismaPg(new Pool({ connectionString, max: 4 })) });
  clients.set(dbName, client);
  return client;
}
```

`src/lib/entities/provision.ts`:
```ts
import { readdirSync, readFileSync } from "node:fs";
import path from "node:path";
import { Pool } from "pg";
import { ENTITY_DB_RE } from "@/lib/entities/naming";
import { entityUrl } from "@/lib/entities/client";
import { assertEntityDatabase } from "@/lib/db-guard";

const MIGRATIONS_DIR = path.join(process.cwd(), "prisma", "entity", "migrations");

type Migration = { version: number; file: string; sql: string };

function migrations(): Migration[] {
  return readdirSync(MIGRATIONS_DIR)
    .filter((f) => /^\d{4}_.+\.sql$/.test(f))
    .sort()
    .map((file) => ({
      version: Number(file.slice(0, 4)),
      file,
      sql: readFileSync(path.join(MIGRATIONS_DIR, file), "utf8"),
    }));
}

export const LATEST_VERSION = Math.max(...migrations().map((m) => m.version));

/** A connection to the server's maintenance database, only to CREATE/DROP. */
function adminPool() {
  const url = new URL(process.env.DATABASE_URL ?? "");
  url.pathname = "/postgres";
  return new Pool({ connectionString: url.toString(), max: 1 });
}

export async function provisionEntityDatabase(dbName: string) {
  if (!ENTITY_DB_RE.test(dbName)) throw new Error(`"${dbName}" is not an entity database name`);
  const admin = adminPool();
  try {
    // Identifiers cannot be parameterised; the regex above is what makes this safe.
    await admin.query(`CREATE DATABASE "${dbName}"`);
  } finally {
    await admin.end();
  }
  const { to } = await migrateEntityDatabase(dbName);
  return { version: to };
}

/** Applies every migration newer than the database's `_entity_migrations` table. */
export async function migrateEntityDatabase(dbName: string) {
  const pool = new Pool({ connectionString: assertEntityDatabase(entityUrl(dbName)), max: 1 });
  try {
    await pool.query(
      `CREATE TABLE IF NOT EXISTS _entity_migrations (version INTEGER PRIMARY KEY, file TEXT NOT NULL, applied_at TIMESTAMPTZ NOT NULL DEFAULT now())`,
    );
    const done = await pool.query<{ version: number }>(`SELECT version FROM _entity_migrations`);
    const have = new Set(done.rows.map((r) => r.version));
    const from = Math.max(0, ...have);
    let to = from;
    for (const m of migrations().filter((x) => !have.has(x.version))) {
      const client = await pool.connect();
      try {
        await client.query("BEGIN");
        await client.query(m.sql);
        await client.query(`INSERT INTO _entity_migrations (version, file) VALUES ($1, $2)`, [m.version, m.file]);
        await client.query("COMMIT");
        to = m.version;
      } catch (error) {
        await client.query("ROLLBACK");
        throw error;
      } finally {
        client.release();
      }
    }
    return { from, to };
  } finally {
    await pool.end();
  }
}

/** Rollback for a creation that failed part-way. Only ever called on a database this attempt created. */
export async function dropFailedEntityDatabase(dbName: string) {
  if (!ENTITY_DB_RE.test(dbName)) throw new Error(`"${dbName}" is not an entity database name`);
  const admin = adminPool();
  try {
    await admin.query(`DROP DATABASE IF EXISTS "${dbName}" WITH (FORCE)`);
  } finally {
    await admin.end();
  }
}
```

```ts
/**
 * What each kind of movement does to a unit or to a stock row.
 *
 * Pure on purpose: the actions in lib/actions/entity-inventory.ts load the
 * current state, call this, and write the movement and the resulting state in
 * one transaction. Keeping the rules here means they are tested without a
 * database, and a screen cannot invent a transition this file refuses.
 */
export type UnitStatus = 'IN_STORAGE' | 'DEPLOYED' | 'RETURNED_TO_OWNER'
export type UnitState = { status: UnitStatus; locationId: string | null }

export type MovementInput = {
  type: 'RECEIVE' | 'MOVE' | 'RELEASE' | 'RETURN'
  fromLocationId?: string | null
  toLocationId?: string | null
  recipient?: string | null
  finalToOwner?: boolean
}

export type Applied = { ok: true; next: UnitState } | { ok: false; reason: string }
const no = (reason: string): { ok: false; reason: string } => ({ ok: false, reason })

export function applyUnitMovement(current: UnitState | null, m: MovementInput): Applied {
  switch (m.type) {
    case 'RECEIVE':
      if (current && current.status !== 'RETURNED_TO_OWNER') return no('That unit is already in custody.')
      if (!m.toLocationId) return no('Choose where it is being stored.')
      return { ok: true, next: { status: 'IN_STORAGE', locationId: m.toLocationId } }
    case 'MOVE':
      if (!current || current.status !== 'IN_STORAGE') return no('Only a unit in storage can be moved. A deployed unit has to be returned first.')
      if (!m.toLocationId) return no('Choose where it is going.')
      if (m.toLocationId === current.locationId) return no('It is already there.')
      return { ok: true, next: { status: 'IN_STORAGE', locationId: m.toLocationId } }
    case 'RELEASE':
      if (!current || current.status !== 'IN_STORAGE') return no('Only a unit in storage can be released.')
      if (!m.recipient?.trim()) return no('Say who it is being released to.')
      if (m.finalToOwner) return { ok: true, next: { status: 'RETURNED_TO_OWNER', locationId: null } }
      if (!m.toLocationId) return no('Choose the destination site.')
      return { ok: true, next: { status: 'DEPLOYED', locationId: m.toLocationId } }
    case 'RETURN':
      if (!current || current.status !== 'DEPLOYED') return no('Only a deployed unit can be returned.')
      if (!m.toLocationId) return no('Choose where it is being stored.')
      return { ok: true, next: { status: 'IN_STORAGE', locationId: m.toLocationId } }
  }
}

export type StockRow = { locationId: string; deployed: boolean; quantity: number }
export type StockApplied = { ok: true; rows: StockRow[] } | { ok: false; reason: string }

function adjust(rows: StockRow[], locationId: string, deployed: boolean, delta: number): StockRow[] | null {
  const next = rows.map((r) => ({ ...r }))
  const row = next.find((r) => r.locationId === locationId && r.deployed === deployed)
  if (!row) {
    if (delta < 0) return null
    next.push({ locationId, deployed, quantity: delta })
    return next
  }
  row.quantity += delta
  if (row.quantity < 0) return null
  return next.filter((r) => r.quantity > 0)
}

export function applyStockMovement(rows: StockRow[], m: MovementInput & { quantity: number }): StockApplied {
  if (!Number.isInteger(m.quantity) || m.quantity <= 0) return no('The quantity has to be a whole number above zero.')
  const q = m.quantity
  let out: StockRow[] | null = rows
  const enough = 'There is not that much there.'

  switch (m.type) {
    case 'RECEIVE':
      if (!m.toLocationId) return no('Choose where it is being stored.')
      out = adjust(rows, m.toLocationId, false, q)
      break
    case 'MOVE':
      if (!m.fromLocationId || !m.toLocationId) return no('Choose where from and where to.')
      if (m.fromLocationId === m.toLocationId) return no('It is already there.')
      out = adjust(rows, m.fromLocationId, false, -q)
      if (!out) return no(enough)
      out = adjust(out, m.toLocationId, false, q)
      break
    case 'RELEASE':
      if (!m.fromLocationId) return no('Choose where it is coming from.')
      if (!m.recipient?.trim()) return no('Say who it is being released to.')
      out = adjust(rows, m.fromLocationId, false, -q)
      if (!out) return no(enough)
      if (!m.finalToOwner) {
        if (!m.toLocationId) return no('Choose the destination site.')
        out = adjust(out, m.toLocationId, true, q)
      }
      break
    case 'RETURN':
      if (!m.fromLocationId || !m.toLocationId) return no('Choose where from and where to.')
      out = adjust(rows, m.fromLocationId, true, -q)
      if (!out) return no(enough)
      out = adjust(out, m.toLocationId, false, q)
      break
  }
  return out ? { ok: true, rows: out } : no(enough)
}
```

```ts
import { auth } from "@/lib/auth";
import { prisma } from "@/lib/prisma";
import { entityDb, type EntityDb } from "@/lib/entities/client";
import { canWriteInEntity, entityIdsFor, type AccessUser, type EntityRole } from "@/lib/entities/access";

export class EntityAccessError extends Error {}

export type EntityContext = {
  entity: { id: string; name: string; slug: string; dbName: string; modules: string[] };
  db: EntityDb;
  userId: string;
  role: EntityRole;
  canWrite: boolean;
};

async function accessUser(userId: string): Promise<AccessUser | null> {
  const user = await prisma.user.findUnique({
    where: { id: userId },
    select: { role: true, entityAccess: true, entityScope: true, disabledAt: true, entityGrants: { select: { entityId: true } } },
  });
  if (!user || user.disabledAt) return null;
  return {
    role: user.role,
    entityAccess: user.entityAccess,
    entityScope: user.entityScope === "ALL" ? "ALL" : "SELECTED",
    grantedEntityIds: user.entityGrants.map((g) => g.entityId),
  };
}

/**
 * The gate every entity page and action goes through. The slug comes from the
 * URL, so it is checked against the grants on every call — a bookmarked link to
 * an entity you have since lost is a refusal, not a page.
 */
export async function requireEntity(slug: string, opts: { write?: boolean } = {}): Promise<EntityContext> {
  const session = await auth();
  if (!session?.user?.id) throw new EntityAccessError("Unauthorized");

  const [user, active] = await Promise.all([
    accessUser(session.user.id),
    prisma.entity.findMany({ where: { active: true }, select: { id: true, name: true, slug: true, dbName: true, modules: true } }),
  ]);
  if (!user) throw new EntityAccessError("Unauthorized");

  const allowed = new Set(entityIdsFor(user, active.map((e) => e.id)));
  const entity = active.find((e) => e.slug === slug && allowed.has(e.id));
  if (!entity) throw new EntityAccessError("You don't have access to that entity.");

  const canWrite = canWriteInEntity(user.role);
  if (opts.write && !canWrite) throw new EntityAccessError("Your role can look at this entity but not change it.");

  return { entity, db: entityDb(entity.dbName), userId: session.user.id, role: user.role, canWrite };
}

/** The entities this user may switch into — for the menu. */
export async function listEntitiesFor(userId: string) {
  const [user, active] = await Promise.all([
    accessUser(userId),
    prisma.entity.findMany({ where: { active: true }, select: { id: true, name: true, slug: true }, orderBy: { name: "asc" } }),
  ]);
  if (!user) return [];
  const allowed = new Set(entityIdsFor(user, active.map((e) => e.id)));
  return active.filter((e) => allowed.has(e.id));
}
```

```ts
export type ItemInput = {
  name: string; description?: string; manufacturer?: string; model?: string;
  category?: string; trackingMode: "SERIALIZED" | "QUANTITY"; notes?: string;

const itemSchema = z.object({ name: z.string().trim().min(1, "Name the item."), trackingMode: z.enum(["SERIALIZED", "QUANTITY"]) });

export async function createItem(slug: string, input: ItemInput): Promise<EntityOutcome> {
  return guarded(slug, async ({ db }) => {

    if (!parsed.success) return fail(parsed.error.issues[0].message);

        model: blank(input.model), notes: blank(input.notes), trackingMode: input.trackingMode,
        ...(category ? { category: { connectOrCreate: { where: { name: category }, create: { name: category } } } } : {}),

    const applied = applyUnitMovement(null, { type: "RECEIVE", toLocationId: input.toLocationId });

          data: {

          },
        });
        await tx.movement.create({
          data: {
            type: "RECEIVE", itemId: item.id, unitId: unit.id, toLocationId: input.toLocationId,
            reference: blank(input.reference), notes: blank(input.notes), actorUserId: userId,

    const applied = applyUnitMovement(
      { status: unit.status, locationId: unit.currentLocationId },
      { type: input.type, toLocationId, fromLocationId: unit.currentLocationId, recipient: input.recipient, finalToOwner: input.finalToOwner },
    );

    await db.$transaction([
      db.movement.create({
        data: {
          type: input.type, itemId: unit.itemId, unitId: unit.id, fromLocationId: unit.currentLocationId, toLocationId,
          recipient: blank(input.recipient), destination: blank(input.destination), reference: blank(input.reference),

          notes: blank(input.notes), actorUserId: userId,

        },
      }),
      db.unit.update({ where: { id: unit.id }, data: { status: applied.next.status, currentLocationId: applied.next.locationId } }),
    ]);
    return done(slug, { MOVE: "Moved.", RELEASE: "Released.", RETURN: "Returned." }[input.type], unit.itemId);
  });

    const applied = applyStockMovement(
      item.stock.map((s) => ({ locationId: s.locationId, deployed: s.deployed, quantity: s.quantity })),
      { type: input.type, quantity: input.quantity, fromLocationId: input.fromLocationId, toLocationId, recipient: input.recipient, finalToOwner: input.finalToOwner },
    );
    if (!applied.ok) return fail(applied.reason);
    await db.$transaction(async (tx) => {
      await tx.movement.create({
        data: {
          type: input.type, itemId: item.id, quantity: input.quantity, fromLocationId: input.fromLocationId ?? null, toLocationId,

          notes: blank(input.notes), actorUserId: userId,
          occurredAt: input.occurredAt ? new Date(input.occurredAt) : undefined,
        },
```

```ts
import { revalidatePath } from "next/cache";
import { prisma } from "@/lib/prisma";
import { requireSuperAdmin } from "@/lib/auth-utils";
import { dbNameFor, slugify } from "@/lib/entities/naming";
import { MODULES } from "@/lib/entities/modules";
import { dropFailedEntityDatabase, provisionEntityDatabase } from "@/lib/entities/provision";

export type EntityFormInput = {
  name: string; contactName?: string; email?: string; phone?: string; address?: string; notes?: string; modules: string[];
};
type Outcome = { status: "ok"; message: string } | { status: "error"; message: string };

const blank = (v?: string) => (v && v.trim() ? v.trim() : null);
const available = new Set(MODULES.filter((m) => m.available).map((m) => m.key as string));

function cleanModules(modules: string[]) {
  return [...new Set(["INVENTORY", ...modules.filter((m) => available.has(m))])];
}

export async function createEntity(
  input: EntityFormInput,
): Promise<{ status: "ok"; id: string; slug: string } | { status: "error"; message: string }> {
  const auth = await requireSuperAdmin();
  if (!auth.authorized) return { status: "error", message: auth.error };

  const name = input.name.trim();
  const slug = slugify(name);
  if (!name || !slug) return { status: "error", message: "Give the entity a name with letters or numbers in it." };
  if (await prisma.entity.findUnique({ where: { slug } })) return { status: "error", message: "An entity with that name already exists." };

  const dbName = dbNameFor(slug);
  let created = false;
  try {
    const { version } = await provisionEntityDatabase(dbName);
    created = true;
    const entity = await prisma.entity.create({
      data: {
        name, slug, dbName, modules: cleanModules(input.modules), schemaVersion: version,
        contactName: blank(input.contactName), email: blank(input.email), phone: blank(input.phone),
        address: blank(input.address), notes: blank(input.notes),
      },
    });
    revalidatePath("/dashboard/settings/entities");
    return { status: "ok", id: entity.id, slug };
  } catch (error) {
    // Only a database this very attempt created is dropped; a name clash means we never created one.
    if (created) await dropFailedEntityDatabase(dbName).catch(() => {});
    console.error(error);
    return { status: "error", message: "The entity's database could not be created, so nothing was saved." };
  }
}

export async function updateEntity(id: string, input: EntityFormInput): Promise<Outcome> {
  const auth = await requireSuperAdmin();
  if (!auth.authorized) return { status: "error", message: auth.error };
  if (!input.name.trim()) return { status: "error", message: "Give the entity a name." };
  // slug and dbName are deliberately not editable: the slug is the database name.
  await prisma.entity.update({
    where: { id },
    data: {
      name: input.name.trim(), modules: cleanModules(input.modules),
      contactName: blank(input.contactName), email: blank(input.email), phone: blank(input.phone),
      address: blank(input.address), notes: blank(input.notes),
    },
  });
  revalidatePath("/dashboard/settings/entities");
  return { status: "ok", message: "Saved." };
}

export async function setEntityActive(id: string, active: boolean): Promise<Outcome> {
  const auth = await requireSuperAdmin();
  if (!auth.authorized) return { status: "error", message: auth.error };
  await prisma.entity.update({ where: { id }, data: { active } });
  revalidatePath("/dashboard/settings/entities");
  return { status: "ok", message: active ? "Reactivated." : "Deactivated. Its data is kept." };
}
```

```ts
    }),

    prisma.userEntity.createMany({ data: valid.map((e) => ({ userId, entityId: e.id })) }),
```

`src/lib/entities/csv.ts`:
```ts
export type ParsedRow = {
  name: string; manufacturer?: string; model?: string; category?: string
  trackingMode: 'SERIALIZED' | 'QUANTITY'; serial?: string; ownerTag?: string; quantity?: number; location: string
}

const REQUIRED = ['name', 'tracking', 'location']

function splitLine(line: string): string[] {
  const out: string[] = []
  let cell = ''
  let quoted = false
  for (let i = 0; i < line.length; i++) {
    const c = line[i]
    if (quoted) {
      if (c === '"' && line[i + 1] === '"') { cell += '"'; i++ }
      else if (c === '"') quoted = false
      else cell += c
    } else if (c === '"') quoted = true
    else if (c === ',') { out.push(cell); cell = '' }
    else cell += c
  }
  out.push(cell)
  return out.map((s) => s.trim())
}

export function parseItemCsv(text: string): { rows: ParsedRow[]; errors: string[] } {
  const lines = text.split(/\r?\n/).filter((l) => l.trim() !== '')
  if (lines.length === 0) return { rows: [], errors: ['The file is empty.'] }
  const header = splitLine(lines[0]).map((h) => h.toLowerCase())
  const missing = REQUIRED.filter((r) => !header.includes(r))
  if (missing.length) return { rows: [], errors: [`Missing column${missing.length > 1 ? 's' : ''}: ${missing.join(', ')}.`] }

  const col = (cells: string[], name: string) => {
    const i = header.indexOf(name)
    const v = i >= 0 ? cells[i] : ''
    return v ? v : undefined
  }
  const rows: ParsedRow[] = []
  const errors: string[] = []
  lines.slice(1).forEach((line, index) => {
    const n = index + 2
    const cells = splitLine(line)
    const name = col(cells, 'name')
    const tracking = (col(cells, 'tracking') ?? '').toUpperCase()
    const location = col(cells, 'location')
    if (!name) return void errors.push(`Line ${n}: name is empty.`)
    if (tracking !== 'SERIALIZED' && tracking !== 'QUANTITY') return void errors.push(`Line ${n}: tracking must be serialized or quantity.`)
    if (!location) return void errors.push(`Line ${n}: location is empty.`)
    const quantityText = col(cells, 'quantity')
    const quantity = quantityText ? Number(quantityText) : undefined
    if (tracking === 'QUANTITY' && (!Number.isInteger(quantity) || (quantity as number) <= 0)) {
      return void errors.push(`Line ${n}: a quantity item needs a whole-number quantity above zero.`)
    }
    rows.push({
      name, manufacturer: col(cells, 'manufacturer'), model: col(cells, 'model'), category: col(cells, 'category'),
      trackingMode: tracking, serial: col(cells, 'serial'), ownerTag: col(cells, 'owner_tag'), quantity, location,
    })
  })
  return { rows, errors }
}
```

```ts
/**
 * Fails if any file outside the entity code imports the entity client, which is
 * the guarantee that entity gear cannot reach VFXnow's depreciation, utilization,
 * map or reports.
 *   npx tsx scripts/check-entity-isolation.ts
 */
import { readdirSync, readFileSync, statSync } from "node:fs";
import path from "node:path";

const ROOT = path.join(process.cwd(), "src");
const ALLOWED = [
  "src/lib/entities/",
  "src/app/entity/",
  "src/components/entities/",
  "src/lib/actions/entity-",
  "src/lib/actions/entities.ts",
  "src/lib/queries/entity-inventory.ts",
  "src/generated/",
];
const NEEDLES = ["@/generated/entity-prisma", "@/lib/entities/client"];

function walk(dir: string): string[] {
  return readdirSync(dir).flatMap((name) => {
    const p = path.join(dir, name);
    return statSync(p).isDirectory() ? walk(p) : /\.(ts|tsx)$/.test(name) ? [p] : [];
  });
}

const offenders = walk(ROOT)
  .map((f) => path.relative(process.cwd(), f))
  .filter((f) => !ALLOWED.some((a) => f.startsWith(a)))
  .filter((f) => NEEDLES.some((n) => readFileSync(f, "utf8").includes(n)));

if (offenders.length) {
  console.error("Entity client imported outside entity code:\n" + offenders.map((f) => "  " + f).join("\n"));
  process.exit(1);
}
console.log("ok   no main-app file imports the entity client");
```
